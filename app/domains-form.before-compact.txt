"use client";

import { useState } from "react";

type Result = {
  domain: string;
  status: string;
  error?: string;
  record?: string;
};

type Action = "add" | "get" | "verify";

export default function DomainsForm() {
  const [domains, setDomains] = useState("");
  const [addResults, setAddResults] = useState<Result[]>([]);
  const [txtRecords, setTxtRecords] = useState<string[]>([]);
  const [txtResults, setTxtResults] = useState<Result[]>([]);
  const [verifyResults, setVerifyResults] = useState<Result[]>([]);
  const [loading, setLoading] = useState("");
  const [progress, setProgress] = useState({ done: 0, total: 0 });
  const [copied, setCopied] = useState(false);

  const getDomains = () =>
    [...new Set(
      domains
        .split(/[\s,;]+/)
        .map(d => d.trim().toLowerCase())
        .filter(Boolean)
    )];

  const list = getDomains();
  const busy = loading !== "";

  const added = addResults.filter(
    r => r.status === "ADDED" || r.status === "ALREADY EXISTS"
  ).length;

  const verified = verifyResults.filter(
    r => r.status === "VERIFIED"
  ).length;

  const failed = [...addResults, ...txtResults, ...verifyResults].filter(
    r => r.status === "FAILED"
  ).length;

  function updateDomains(value: string) {
    setDomains(value);
    setAddResults([]);
    setTxtRecords([]);
    setTxtResults([]);
    setVerifyResults([]);
    setCopied(false);
  }

  async function request(domain: string, action: Action): Promise<Result> {
    const response = await fetch(
      action === "add" ? "/api/domains" : "/api/verify",
      {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(
          action === "add" ? { domain } : { domain, action }
        ),
      }
    );

    const data = await response.json();

    return {
      domain,
      status: data.status || "FAILED",
      error: data.error || (!response.ok ? "Request failed" : undefined),
      record: data.record,
    };
  }

  async function run(action: Action) {
    if (busy || !list.length) return;

    setLoading(action);
    setProgress({ done: 0, total: list.length });
    setCopied(false);

    if (action === "add") { setAddResults([]); setTxtRecords([]); setTxtResults([]); }
    if (action === "get") {
      setTxtRecords([]);
      setTxtResults([]);
    }
    if (action === "verify") setVerifyResults([]);

    for (let i = 0; i < list.length; i++) {
      const domain = list[i];
      let result: Result;

      try {
        result = await request(domain, action);
      } catch {
        result = {
          domain,
          status: "FAILED",
          error: "Connection error",
        };
      }

      if (action === "add") {
        setAddResults(prev => [...prev, result]);

        if (result.status === "ADDED" || result.status === "ALREADY EXISTS") {
          try {
            const txt = await request(domain, "get");

            if (txt.record) {
              setTxtRecords(prev => [...prev, txt.record!]);
            }

            setTxtResults(prev => [...prev, txt]);
          } catch {
            setTxtResults(prev => [
              ...prev,
              { domain, status: "FAILED", error: "TXT extraction failed" }
            ]);
          }
        }
      }

      if (action === "get") {
        if (result.record) {
          setTxtRecords(prev => [...prev, result.record!]);
        }
        setTxtResults(prev => [...prev, result]);
      }

      if (action === "verify") {
        setVerifyResults(prev => [...prev, result]);
      }

      setProgress({ done: i + 1, total: list.length });
    }

    setLoading("");
  }

  const exportText = txtRecords.join("\n");

  async function copyRecords() {
    try {
      await navigator.clipboard.writeText(exportText);
      setCopied(true);
    } catch {
      setCopied(false);
      alert("Copy failed. Please select and copy the records manually.");
    }
  }

  function statusColor(status: string) {
    if (status === "ADDED" || status === "VERIFIED" ||
        status === "TXT READY") {
      return "bg-emerald-500/10 text-emerald-300 border-emerald-500/20";
    }
    if (status === "FAILED") {
      return "bg-rose-500/10 text-rose-300 border-rose-500/20";
    }
    return "bg-amber-500/10 text-amber-300 border-amber-500/20";
  }

  function Results({
    title,
    data,
  }: {
    title: string;
    data: Result[];
  }) {
    if (!data.length) return null;

    return (
      <div className="space-y-3">
        <h3 className="text-sm font-semibold text-slate-200">{title}</h3>
        <div className="overflow-hidden rounded-2xl border border-white/10">
          {data.map((r, i) => (
            <div
              key={`${r.domain}-${i}`}
              className="flex flex-col gap-2 border-b border-white/5 bg-white/[0.03] p-4 last:border-b-0 sm:flex-row sm:items-center sm:justify-between"
            >
              <div className="min-w-0">
                <p className="break-all text-sm font-medium text-white">
                  {r.domain}
                </p>
                {r.error && (
                  <p className="mt-1 break-words text-xs text-rose-300">
                    {r.error}
                  </p>
                )}
              </div>
              <span className={`w-fit shrink-0 rounded-full border px-3 py-1 text-xs font-semibold ${statusColor(r.status)}`}>
                {r.status}
              </span>
            </div>
          ))}
        </div>
      </div>
    );
  }

  return (
    <div className="space-y-3">

      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
        {[
          { label: "Total Domains", value: list.length, color: "text-blue-300" },
          { label: "Added", value: added, color: "text-emerald-300" },
          { label: "TXT Ready", value: txtRecords.length, color: "text-violet-300" },
          { label: "Verified", value: verified, color: "text-cyan-300" },
        ].map(item => (
          <div
            key={item.label}
            className="rounded-2xl border border-white/10 bg-white/[0.04] p-3"
          >
            <p className="text-xs font-medium text-slate-400">
              {item.label}
            </p>
            <p className={`mt-3 text-3xl font-bold ${item.color}`}>
              {item.value}
            </p>
          </div>
        ))}
      </div>

      <section className="rounded-2xl border border-white/10 bg-slate-950/40 p-3">
        <div className="mb-2 flex flex-wrap items-center justify-between gap-3">
          <div>
            <p className="text-xs font-semibold uppercase tracking-[0.2em] text-blue-300">
              Step 01
            </p>
            <h2 className="mt-2 text-xl font-bold text-white">
              Domain Management
            </h2>
            <p className="mt-1 text-sm text-slate-400">
              Paste one or multiple domains to manage them in bulk.
            </p>
          </div>
          <span className="rounded-full border border-blue-500/20 bg-blue-500/10 px-4 py-2 text-xs font-semibold text-blue-300">
            {list.length} Domains
          </span>
        </div>

        <textarea
          value={domains}
          onChange={e => updateDomains(e.target.value)}
          disabled={busy}
          placeholder={"example1.com\nexample2.com\nexample3.com"}
          rows={3}
          spellCheck={false}
          className="w-full resize-y rounded-xl border border-white/10 bg-slate-900/80 p-4 font-mono text-sm leading-7 text-white outline-none transition placeholder:text-slate-600 focus:border-blue-500 focus:ring-2 focus:ring-blue-500/10 disabled:opacity-60"
        />

        <div className="mt-2 grid gap-2 sm:grid-cols-2">
          <button
            onClick={() => run("add")}
            disabled={busy || !list.length}
            className="rounded-xl bg-blue-600 px-5 py-3.5 text-sm font-bold text-white shadow-lg shadow-blue-600/15 transition hover:bg-blue-500 disabled:cursor-not-allowed disabled:opacity-40"
          >
            {loading === "add" ? "ADDING DOMAINS..." : "+ ADD DOMAINS"}
          </button>

          <button
            onClick={() => run("get")}
            disabled={busy || !list.length}
            className="rounded-xl border border-violet-400/30 bg-violet-500/15 px-5 py-3.5 text-sm font-bold text-violet-200 transition hover:bg-violet-500/25 disabled:cursor-not-allowed disabled:opacity-40"
          >
            {loading === "get" ? "FETCHING RECORDS..." : "GET TXT RECORDS"}
          </button>
        </div>
      </section>

      {busy && (
        <div className="rounded-2xl border border-blue-400/20 bg-blue-500/5 p-5">
          <div className="mb-3 flex justify-between text-sm">
            <span className="font-medium text-blue-200">
              Processing domains...
            </span>
            <span className="text-slate-300">
              {progress.done} / {progress.total}
            </span>
          </div>
          <div className="h-2 overflow-hidden rounded-full bg-slate-800">
            <div
              className="h-full rounded-full bg-blue-500 transition-all"
              style={{
                width: `${progress.total ? progress.done / progress.total * 100 : 0}%`
              }}
            />
          </div>
        </div>
      )}

      <Results title="Add Domain Results" data={addResults} />

      <section className="rounded-2xl border border-white/10 bg-slate-950/40 p-3">
        <div className="mb-2 flex flex-wrap items-center justify-between gap-3">
          <div>
            <p className="text-xs font-semibold uppercase tracking-[0.2em] text-violet-300">
              Step 02
            </p>
            <h2 className="mt-2 text-xl font-bold text-white">
              TXT Records Export
            </h2>
            <p className="mt-1 text-sm text-slate-400">
              Copy DNS verification records for your external DNS app.
            </p>
          </div>
          <span className="rounded-full border border-violet-500/20 bg-violet-500/10 px-4 py-2 text-xs font-semibold text-violet-300">
            {txtRecords.length} Records
          </span>
        </div>

        

        <button
          onClick={copyRecords}
          disabled={!txtRecords.length}
          className="w-full rounded-xl bg-emerald-600 px-5 py-3.5 text-sm font-bold text-white transition hover:bg-emerald-500 disabled:cursor-not-allowed disabled:opacity-40"
        >
          {copied ? "COPIED SUCCESSFULLY" : "COPY ALL RECORDS"}
        </button>
      </section>

      

      <section className="rounded-2xl border border-white/10 bg-slate-950/40 p-3">
        <div className="flex flex-col justify-between gap-5 sm:flex-row sm:items-center">
          <div>
            <p className="text-xs font-semibold uppercase tracking-[0.2em] text-cyan-300">
              Step 03
            </p>
            <h2 className="mt-2 text-xl font-bold text-white">
              DNS Verification
            </h2>
            <p className="mt-1 max-w-lg text-sm text-slate-400">
              After publishing TXT records in DNS, request domain verification.
            </p>
          </div>

          <button
            onClick={() => run("verify")}
            disabled={busy || !list.length}
            className="shrink-0 rounded-xl border border-cyan-400/30 bg-cyan-500/15 px-7 py-3.5 text-sm font-bold text-cyan-200 transition hover:bg-cyan-500/25 disabled:cursor-not-allowed disabled:opacity-40"
          >
            {loading === "verify" ? "VERIFYING..." : "VERIFY ALL"}
          </button>
        </div>
      </section>

      <Results title="Verification Results" data={verifyResults} />

      {failed > 0 && (
        <p className="text-center text-xs text-rose-300">
          {failed} failed operation(s). Check the results above for details.
        </p>
      )}

      <div className="border-t border-white/10 pt-5 text-center">
        <p className="text-xs text-slate-500">
          Postmaster Manager · Created by Ilyas Houari
        </p>
      </div>
    </div>
  );
}
